cmake_minimum_required(VERSION 3.20)
project(coro_os LANGUAGES CXX)
option(CORO_OS_TESTS "Build host tests" ON)
option(CORO_OS_SANITIZERS "Enable host ASan and UBSan" OFF)
set(NOALLOC_CXX "" CACHE FILEPATH "Compiler supporting coro::noalloc and frame builtins")

add_library(coro_os INTERFACE)
target_include_directories(coro_os INTERFACE include)
target_compile_features(coro_os INTERFACE cxx_std_20)
if(CORO_OS_SANITIZERS)
  target_compile_options(coro_os INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(coro_os INTERFACE -fsanitize=address,undefined)
endif()

if(CORO_OS_TESTS)
  enable_testing()
  foreach(test queues scheduler notification lifecycle idle template_port singleton sync nucleo_behavior)
    add_executable(${test}_test tests/${test}_test.cpp)
    target_link_libraries(${test}_test PRIVATE coro_os)
    target_compile_options(${test}_test PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
    add_test(NAME ${test} COMMAND ${test}_test)
  endforeach()
  target_sources(lifecycle_test PRIVATE tests/support/allocation_probe.cpp)
  target_sources(singleton_test PRIVATE tests/singleton_other_tu.cpp)
  find_program(BASH_EXECUTABLE bash)
  if(BASH_EXECUTABLE)
    add_test(NAME integration_scripts COMMAND ${BASH_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/scripts_test.sh)
    add_test(NAME noalloc_integration COMMAND ${CMAKE_COMMAND} -E env
      "NOALLOC_CXX=${NOALLOC_CXX}"
      ${BASH_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/check_noalloc.sh)
    set_tests_properties(noalloc_integration PROPERTIES SKIP_RETURN_CODE 77)
  endif()
endif()
